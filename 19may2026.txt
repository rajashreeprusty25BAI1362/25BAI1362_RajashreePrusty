CREATE TABLE Worker (
    WORKER_ID INTEGER PRIMARY KEY AUTOINCREMENT,
    FIRST_NAME TEXT,
    LAST_NAME TEXT,
    SALARY INTEGER,
    JOINING_DATE DATETIME,
    DEPARTMENT TEXT
);

SELECT * 
FROM Worker 
WHERE FIRST_NAME LIKE '_i__a_';

SELECT SALARY 
FROM Worker 
WHERE DEPARTMENT = 'Admin';

SELECT SALARY FROM Worker
UNION ALL
SELECT SALARY FROM Worker;

SELECT WORKER_ID AS RollNumber 
FROM Worker;

SELECT DISTINCT DEPARTMENT 
FROM Worker;

SELECT WORKER_ID, FIRST_NAME,
CASE
    WHEN SALARY > 300000 THEN 'Rich People'
    WHEN SALARY >= 100000 AND SALARY < 300000 THEN 'Middle Class'
    ELSE 'Poor'
END AS Status
FROM Worker;

SELECT MIN(SALARY) AS MinimumSalary
FROM Worker
WHERE DEPARTMENT = 'Admin';

SELECT MAX(SALARY) AS MaximumSalary
FROM Worker
WHERE DEPARTMENT = 'Admin';

CREATE VIEW MinSalaryAdmin AS
SELECT MIN(SALARY) AS MinSalary
FROM Worker
WHERE DEPARTMENT = 'Admin';

CREATE VIEW MaxSalaryAdmin AS
SELECT MAX(SALARY) AS MaxSalary
FROM Worker
WHERE DEPARTMENT = 'Admin';

SELECT * FROM MaxSalaryAdmin;

SELECT FIRST_NAME, SALARY,
SALARY + 5000 AS Updated_Salary
FROM Worker;

SELECT FIRST_NAME, SALARY,
SALARY - 2000 AS Salary_After_Tax
FROM Worker;

SELECT FIRST_NAME, SALARY,
SALARY * 12 AS Annual_Salary
FROM Worker;

SELECT FIRST_NAME, SALARY,
SALARY / 12 AS Monthly_Salary
FROM Worker;

SELECT FIRST_NAME, SALARY,
SALARY % 2 AS Salary_Remainder
FROM Worker;
