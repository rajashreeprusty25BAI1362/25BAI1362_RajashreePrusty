    cid INT PRIMARY KEY,
    cname VARCHAR(100) NOT NULL
);

CREATE TABLE Product (
    pid INT PRIMARY KEY,
    pname VARCHAR(100) NOT NULL,
    cid INT,

    -- Foreign key with delete and update cascade
    FOREIGN KEY (cid)
    REFERENCES Category(cid)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);


DESC Product;
