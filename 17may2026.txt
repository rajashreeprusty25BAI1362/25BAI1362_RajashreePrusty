-- Equal To (=)
SELECT * FROM Worker
WHERE SALARY = 105000;

-- Greater Than (>)
SELECT * FROM Worker
WHERE SALARY > 100000;

-- Less Than (<)
SELECT * FROM Worker
WHERE SALARY < 100000;

-- Greater Than or Equal To (>=)
SELECT * FROM Worker
WHERE SALARY >= 200000;

-- Less Than or Equal To (<=)
SELECT * FROM Worker
WHERE SALARY <= 100000;

-- Not Equal To (!= or <>)
SELECT * FROM Worker
WHERE DEPARTMENT != 'HR';

-- BETWEEN Comparator
SELECT * FROM Worker
WHERE SALARY BETWEEN 80000 AND 300000;

-- IN Comparator
SELECT * FROM Worker
WHERE DEPARTMENT IN ('HR', 'Admin');

-- LIKE Comparator
SELECT * FROM Worker
WHERE FIRST_NAME LIKE 'V%';

-- IS NULL Comparator
SELECT * FROM CSE
WHERE sLocation IS NULL;
