USE vitcampus;

-- Table creation with NOT NULL and UNIQUE constraints
CREATE TABLE Persons (
    ID INT NOT NULL,
    LastName VARCHAR(255) UNIQUE NOT NULL,
    FirstName VARCHAR(255) NOT NULL,
    Age INT
);

-- Display table structure
DESC Persons;

-- Insert a record
INSERT INTO Persons
VALUES (101, 'V', 'Jayanth', NULL);

-- Display records
SELECT * FROM Persons;

-- Add UNIQUE constraint to Age column
ALTER TABLE Persons
ADD UNIQUE (Age);

-- Remove constraint
ALTER TABLE Persons
DROP CONSTRAINT Age;

-- Add primary key to ID column
ALTER TABLE Persons
ADD PRIMARY KEY (ID);

-- Remove primary key
ALTER TABLE Persons
DROP PRIMARY KEY;

-- Remove table
DROP TABLE Persons;


-- CHECK constraint example
CREATE TABLE Persons2 (
    Age INT NOT NULL,

    -- Age should be between 18 and 23
    CHECK (Age >= 18 AND Age <= 23)
);

-- Insert valid data
INSERT INTO Persons2
VALUES (21);

SELECT * FROM Persons2;


-- DEFAULT constraint example
CREATE TABLE Persons1 (
    ID INT NOT NULL,
    LastName VARCHAR(255) NOT NULL,
    FirstName VARCHAR(255),
    Age INT,

    -- Default city value
    City VARCHAR(255) DEFAULT 'Coimbatore'
);

-- Display table structure
DESC Persons1;

-- Insert record without city
INSERT INTO Persons1 (ID, LastName, FirstName, Age)
VALUES (201, 'Kumar', 'Rahul', 20);

SELECT * FROM Persons1;


-- PRIMARY KEY and UNIQUE example
CREATE TABLE Products (
    pid INT PRIMARY KEY,
    pname VARCHAR(100) NOT NULL,

    -- Preference number should be unique
    prefNo INT NOT NULL UNIQUE
);

DESC Products;

INSERT INTO Products
VALUES (102, 'Furniture', 19930);

SELECT * FROM Products;

DROP TABLE Products;


-- Parent table creation
CREATE TABLE Category (
    cid INT PRIMARY KEY,
    cname VARCHAR(100) NOT NULL
);

-- Insert category records
INSERT INTO Category
VALUES
(101, 'Electronics'),
(102, 'Home Appliances');

SELECT * FROM Category;


-- FOREIGN KEY example
CREATE TABLE Products_New (
    pid INT PRIMARY KEY,
    pname VARCHAR(100) NOT NULL,
    cid INT,

    -- Link category ID with Category table
    FOREIGN KEY (cid)
    REFERENCES Category(cid)
);

DESC Products_New;

-- Insert valid foreign key data
INSERT INTO Products_New
VALUES (504, 'Power Bank Samsung', 101);

SELECT * FROM Products_New;

-- Delete operation
-- This may fail if the category is referenced
DELETE FROM Category
WHERE cid = 101;

-- Update record
UPDATE Products_New
SET pname = 'Samsung Power Bank'
WHERE pid = 504;

SELECT * FROM Products_New;
